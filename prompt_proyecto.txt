Future<void> main() async {
  WidgetsFlutterBinding.ensureInitialized();

  // Persistencia local (Hive) - 100% offline
  await HiveConfig.init();
  await HiveService.registerAdapters();
  await HiveService.initBoxes();
  await HiveService.seedIfEmpty();
  if (HiveService.userBox.isEmpty) {
    await HiveService.saveUser(UserModel.initial());
  }

  // Cámara disponible en el dispositivo (si falla, la app sigue navegable)
  List<CameraDescription> cameras;
  try {
    cameras = await availableCameras();
  } catch (_) {
    cameras = <CameraDescription>[];
  }

  runApp(ProviderScope(child: MyApp(cameras: cameras)));
}

import 'dart:async';

import 'package:camera/camera.dart';
import 'package:flutter/material.dart';
import 'package:logger/logger.dart';
import 'package:senasgo/core/constants/app_constants.dart';

class CameraService extends ChangeNotifier {
  CameraService(this.cameras);

  final List<CameraDescription> cameras;
  CameraController? _controller;
  bool _isInitialized = false;
  bool _isDetecting = false;
  bool _hasError = false;
  String _errorMessage = '';
  int _lastFrameTime = 0;
  final Logger _logger = Logger();

  CameraController? get controller => _controller;
  bool get isInitialized => _isInitialized;
  bool get isDetecting => _isDetecting;
  bool get hasError => _hasError;
  String get errorMessage => _errorMessage;

  Future<void> initialize() async {
    try {
      if (cameras.isEmpty) {
        _hasError = true;
        _errorMessage = 'No se encontraron cámaras disponibles';
        notifyListeners();
        return;
      }

      _controller = CameraController(
        cameras.firstWhere(
          (c) => c.lensDirection == CameraLensDirection.front,
          orElse: () => cameras.first,
        ),
        ResolutionPreset.medium,
        enableAudio: false,
        imageFormatGroup: ImageFormatGroup.yuv420,
      );

      await _controller!.initialize();

      // Configurar para optimizar rendimiento y evitar sobrecalentamiento
      await _controller!.setFocusMode(FocusMode.auto);
      await _controller!.setExposureMode(ExposureMode.auto);

      _isInitialized = true;
      _hasError = false;
      notifyListeners();
    } catch (e) {
      _logger.e('Error al inicializar cámara: $e');
      _hasError = true;
      _errorMessage = 'Error al acceder a la cámara: $e';
      notifyListeners();
    }
  }

  void startImageStream(void Function(CameraImage image) onImageAvailable) {
    if (_controller == null || !_controller!.value.isInitialized) {
      return;
    }

    _isDetecting = true;
    _controller!.startImageStream((image) {
      final currentTime = DateTime.now().millisecondsSinceEpoch;
      // Limitar a 10-15 FPS (Throttle)
      if (currentTime - _lastFrameTime < 1000 ~/ AppConstants.targetFpsMax) {
        return;
      }
      _lastFrameTime = currentTime;
      onImageAvailable(image);
    });
    notifyListeners();
  }

  void stopImageStream() {
    if (_controller != null && _controller!.value.isStreamingImages) {
      _controller!.stopImageStream();
    }
    _isDetecting = false;
    notifyListeners();
  }

  Future<void> disposeController() async {
    stopImageStream();
    await _controller?.dispose();
    _controller = null;
    _isInitialized = false;
    notifyListeners();
  }

  @override
  void dispose() {
    disposeController();
    super.dispose();
  }
}

import 'dart:io';

import 'package:flutter/foundation.dart';

/// Detecta la capacidad del dispositivo para decidir el nivel de detalle
/// del avatar del profesor (Requerimiento: Fallback Visual por poca RAM).
class DevicePerformance {
  DevicePerformance._();

  static bool? _cachedLowMemory;

  /// Heurística de memoria disponible:
  /// 1. En móvil, se lee RSS del proceso (resident set).
  /// 2. Se considera "poca RAM" si el proceso supera el umbral o si el
  ///    sistema reporta un perfil de gama baja.
  static bool get isLowMemory {
    if (kIsWeb) return false;
    if (_cachedLowMemory != null) return _cachedLowMemory!;

    var low = false;
    try {
      if (Platform.isAndroid || Platform.isIOS) {
        // RSS aproximado en bytes del proceso de la app
        final rss = ProcessInfo.currentRss; // resident set size
        const lowRamThreshold = 300 * 1024 * 1024; // 300 MB de RSS
        low = rss > lowRamThreshold;
      }
    } catch (_) {
      // Si no se puede medir, se asume capacidad suficiente
      low = false;
    }

    _cachedLowMemory = low;
    return low;
  }

  /// FPS efectivo permitido (control térmico 10-15 FPS).
  static int get targetFps => isLowMemory ? 10 : 15;
}

import 'package:hive/hive.dart';
import 'package:senasgo/core/constants/app_constants.dart';
import 'package:senasgo/features/learning/data/seed_data.dart';
import 'package:senasgo/shared/models/lesson_model.dart';

import 'package:senasgo/shared/models/sign_model.dart';
import 'package:senasgo/shared/models/store_item_model.dart';
import 'package:senasgo/shared/models/user_model.dart';

/// Acceso único a la persistencia local (Hive). 100% offline.
class HiveService {
  static Future<void> registerAdapters() async {
    if (!Hive.isAdapterRegistered(0)) Hive.registerAdapter(UserModelAdapter());
    if (!Hive.isAdapterRegistered(1)) {
      Hive.registerAdapter(LessonModelAdapter());
    }
    if (!Hive.isAdapterRegistered(2)) {
      Hive.registerAdapter(SignCategoryAdapter());
    }
    if (!Hive.isAdapterRegistered(3)) Hive.registerAdapter(SignModelAdapter());
    if (!Hive.isAdapterRegistered(4)) {
      Hive.registerAdapter(ProgressModelAdapter());
    }
    if (!Hive.isAdapterRegistered(5)) {
      Hive.registerAdapter(StoreItemTypeAdapter());
    }
    if (!Hive.isAdapterRegistered(6)) {
      Hive.registerAdapter(StoreItemModelAdapter());
    }
  }

  static Future<void> initBoxes() async {
    await Hive.openBox<UserModel>(AppConstants.hiveBoxUser);
    await Hive.openBox<LessonModel>(AppConstants.hiveBoxLessons);
    await Hive.openBox<SignModel>(AppConstants.hiveBoxVocabulary);
    await Hive.openBox<ProgressModel>(AppConstants.hiveBoxProgress);
    await Hive.openBox<StoreItemModel>(AppConstants.hiveBoxStore);
  }

  /// Inserta el vocabulario acotado y la tienda solo la primera vez.
  static Future<void> seedIfEmpty() async {
    if (lessonsBox.isEmpty) {
      for (final lesson in SeedData.lessons) {
        await lessonsBox.put(lesson.id, lesson);
      }
    }
    if (vocabularyBox.isEmpty) {
      for (final sign in SeedData.signs) {
        await vocabularyBox.put(sign.id, sign);
      }
    }
    if (storeBox.isEmpty) {
      for (final item in SeedData.storeItems) {
        await storeBox.put(item.id, item);
      }
    }
  }

  // ─────────────────────────── Boxes ───────────────────────────
  static Box<UserModel> get userBox =>
      Hive.box<UserModel>(AppConstants.hiveBoxUser);
  static Box<LessonModel> get lessonsBox =>
      Hive.box<LessonModel>(AppConstants.hiveBoxLessons);
  static Box<SignModel> get vocabularyBox =>
      Hive.box<SignModel>(AppConstants.hiveBoxVocabulary);
  static Box<ProgressModel> get progressBox =>
      Hive.box<ProgressModel>(AppConstants.hiveBoxProgress);
  static Box<StoreItemModel> get storeBox =>
      Hive.box<StoreItemModel>(AppConstants.hiveBoxStore);

  // ─────────────────────────── Usuario ───────────────────────────
  static UserModel get currentUser => userBox.values.isEmpty
      ? UserModel.initial()
      : userBox.values.first;

  static Future<void> saveUser(UserModel user) async {
    await userBox.clear();
    await userBox.put(user.id, user);
  }

  /// Otorga XP locales (ciclo: Practicar -> Recibir Feedback).
  static Future<UserModel> addXp(int amount) async {
    final user = currentUser;
    final newXp = user.xp + amount;
    final updated = user.copyWith(
      xp: newXp,
      level: calculateLevel(newXp),
      lastLogin: DateTime.now(),
    );
    await saveUser(updated);
    return updated;
  }

  /// Nivel dinámico: a mayor nivel, más estricto el umbral de IA.
  static int calculateLevel(int xp) {
    if (xp < 100) return 1;
    if (xp < 300) return 2;
    if (xp < 600) return 3;
    if (xp < 1000) return 4;
    return 5;
  }

  // ─────────────────────────── Lecciones ───────────────────────────
  static List<LessonModel> get lessons {
    final list = lessonsBox.values.toList()
      ..sort((a, b) => a.order.compareTo(b.order));
    return list;
  }

  static Future<void> markLessonCompleted(String lessonId) async {
    final lesson = lessonsBox.get(lessonId);
    if (lesson == null || lesson.isCompleted) return;
    await lessonsBox.put(lessonId, lesson.copyWith(isCompleted: true));

    final user = currentUser;
    if (!user.completedLessons.contains(lessonId)) {
      await saveUser(user.copyWith(
        completedLessons: [...user.completedLessons, lessonId],
      ));
    }
  }

  static SignModel? signById(String id) => vocabularyBox.get(id);

  static List<SignModel> signsOf(String lessonId) {
    final lesson = lessonsBox.get(lessonId);
    if (lesson == null) return const [];
    return lesson.signIds
        .map((id) => vocabularyBox.get(id))
        .whereType<SignModel>()
        .toList();
  }

  // ─────────────────────────── Tienda ───────────────────────────
  static List<StoreItemModel> get storeItems => storeBox.values.toList();

  /// Compra con XP locales. Devuelve null si no alcanza.
  static Future<StoreItemModel?> buyItem(String itemId) async {
    final item = storeBox.get(itemId);
    if (item == null || item.isUnlocked) return null;

    final user = currentUser;
    if (user.xp < item.priceXp) return null;

    final updatedUser = user.copyWith(xp: user.xp - item.priceXp);
    final unlocked = item.copyWith(isUnlocked: true, isEquipped: true);
    await saveUser(updatedUser);
    await storeBox.put(itemId, unlocked);
    return unlocked;
  }

  /// Equipa/desequipa un ítem (solo uno por tipo).
  static Future<StoreItemModel?> toggleEquip(String itemId) async {
    final item = storeBox.get(itemId);
    if (item == null || !item.isUnlocked) return null;

    if (item.isEquipped) {
      final off = item.copyWith(isEquipped: false);
      await storeBox.put(itemId, off);
      return off;
    }

    // Un solo ítem equipado por tipo
    for (final other in storeBox.values) {
      if (other.type == item.type && other.isEquipped) {
        await storeBox.put(other.id, other.copyWith(isEquipped: false));
      }
    }
    final on = item.copyWith(isEquipped: true);
    await storeBox.put(itemId, on);
    return on;
  }

  static StoreItemModel? equippedOf(StoreItemType type) {
    for (final item in storeBox.values) {
      if (item.type == type && item.isEquipped) return item;
    }
    return null;
  }
}

import 'dart:async';
import 'dart:math';

import 'package:camera/camera.dart';

class InferenceResult {
  const InferenceResult({
    required this.isCorrect,
    required this.confidence,
    required this.feedback,
    required this.parameters,
  });

  final bool isCorrect;
  final double confidence;
  final String feedback;
  final Map<String, double> parameters; // manual, palm, place, movement, face
}

class InferenceService extends ChangeNotifier {
  Interpreter? _interpreter;
  bool _isLoaded = false;
  bool _isBusy = false;
  final Logger _logger = Logger();

  bool get isLoaded => _isLoaded;
  bool get isBusy => _isBusy;

  Future<void> loadModel() async {
    try {
      _interpreter = await Interpreter.fromAsset('assets/models/gesture_model.tflite');
      _isLoaded = true;
      notifyListeners();
    } catch (e) {
      _logger.e('Error cargando modelo TFLite: $e');
      _isLoaded = false;
      notifyListeners();
    }
  }

  double getDynamicThreshold(int userLevel) {
    // Umbral dinámico: principiantes más tolerantes
    final threshold = AppConstants.levelThresholds[userLevel] ??
        AppConstants.levelThresholds[1]!;
    return threshold;
  }

  Future<InferenceResult?> processFrame({
    required CameraImage image,
    required String targetLabel,
    required int userLevel,
  }) async {
    if (!_isLoaded || _isBusy || _interpreter == null) {
      return null;
    }

    _isBusy = true;

    try {
      // TODO: Implementar preprocesamiento real con MediaPipe Hands + TFLite
      // Esta implementación es el scaffold para producción
      await Future<void>.delayed(const Duration(milliseconds: 30));

      // Simulación para desarrollo (reemplazar con inferencia real)
      final random = Random();
      final confidence = 0.4 + random.nextDouble() * 0.5; // 0.4 - 0.9
      final threshold = getDynamicThreshold(userLevel);

      final params = {
        'manualConfig': 0.7 + random.nextDouble() * 0.3,
        'palmOrientation': 0.7 + random.nextDouble() * 0.3,
        'articulationPlace': 0.7 + random.nextDouble() * 0.3,
        'movement': 0.7 + random.nextDouble() * 0.3,
        'facialExpression': 0.6 + random.nextDouble() * 0.3,
      };

      String feedback = '✕ Necesita mejorar: La posición de la mano no coincide con la guía';
      bool isCorrect = false;

      if (confidence >= threshold) {
        // Verificar todos los parámetros formativos (>=0.6 heurístico)
        final allOk = params.values.every((v) => v >= 0.6);
        if (allOk) {
          isCorrect = true;
          feedback = '✓ ¡Excelente! Postura correcta';
        } else {
          // Identificar qué mejorar
          final issues = <String>[];
          if (params['manualConfig']! < 0.6) {
            issues.add('la configuración manual');
          }
          if (params['palmOrientation']! < 0.6) {
            issues.add('la orientación de la palma');
          }
          if (params['articulationPlace']! < 0.6) {
            issues.add('el lugar de articulación');
          }
          if (params['movement']! < 0.6) {
            issues.add('el movimiento');
          }
          if (params['facialExpression']! < 0.6) {
            issues.add('la expresión facial');
          }
          if (issues.isNotEmpty) {
            feedback =
                '✕ Necesita mejorar: ${issues.join(', ')}. Revisa la guía.';
          }
        }
      }

      return InferenceResult(
        isCorrect: isCorrect,
        confidence: confidence,
        feedback: feedback,
        parameters: params,
      );
    } catch (e) {
      _logger.e('Error procesando frame: $e');
      return null;
    } finally {
      _isBusy = false;
    }
  }

  Future<bool> checkCameraPermission() async {
    final status = await Permission.camera.status;
    if (status.isDenied) {
      final result = await Permission.camera.request();
      return result.isGranted;
    }
    return status.isGranted;
  }

  Future<bool> checkEnvironment({
    required CameraController controller,
    required CameraImage image,
  }) async {
    // TODO: Implementar verificación de iluminación real
    // Contraluz fuerte o ambiente oscuro
    await Future<void>.delayed(const Duration(milliseconds: 10));

  }

  @override
  void dispose() {
    _interpreter?.close();
    super.dispose();

/// Los 11 estados del módulo de práctica (según documentación v3_04):
/// preparación -> encuadre -> evaluación -> feedback -> siguiente/resumen,
/// más estados de contingencia (permisos, iluminación, sin mano, error).
enum PracticePhase {
  preparing, // 1. Prepararse: fase obligatoria de encuadre
  detecting, // 2. Captura: buscando la mano
  evaluating, // 3. Evaluación de la postura
  feedback, // 4. Recibir feedback (correcto/incorrecto)
  completed, // 5. Seña correcta (+10 XP)
  lessonFinished, // Resumen de lección (XP + insignia)
  error, // Contingencia: permiso/cámara/sensor
}

class PracticeState {
  const PracticeState({

    required this.currentSignId,

    required this.totalSigns,
    required this.feedbackMessage,
    required this.isPositiveFeedback,
    required this.showGuide,

    required this.user,
  });

  final PracticePhase phase;
  final bool handDetected;
  final bool lightingOk;
  final String currentSignId;

  final int totalSigns;
  final String feedbackMessage;
  final bool isPositiveFeedback;

class PracticeController extends ChangeNotifier {
  PracticeController({
    required this.cameraService,
    required this.inferenceService,
  });

  final CameraService cameraService;
  final InferenceService inferenceService;
  final Logger _logger = Logger();

  List<SignModel> _signs = const [];
  int _signIndex = 0;
  bool _isProcessing = false;
  Timer? _feedbackTimer;

  PracticeState _state = PracticeState(
    phase: PracticePhase.preparing,
    handDetected: false,
    lightingOk: true,
    currentSignId: '',
    signName: '',
    attempts: 0,
    correctCount: 0,
    totalSigns: 0,
    feedbackMessage: 'Coloca tu mano dentro del área indicada.',
    isPositiveFeedback: false,
    showGuide: true,
    showSummary: false,
    user: HiveService.currentUser,
  );

  PracticeState get state => _state;

  /// Carga la lección elegida y arranca el ciclo Aprender->...->Corregir.
  Future<void> startLesson(String? lessonId) async {
    if (lessonId != null) {
      _signs = HiveService.signsOf(lessonId);
    }
    if (_signs.isEmpty) {
      // Práctica libre con todo el vocabulario
      _signs = HiveService.vocabularyBox.values.toList();
    }
    _signIndex = 0;

    final first = _signs.isEmpty ? null : _signs.first;
    _state = _state.copyWith(
      currentSignId: first?.id ?? '',
      signName: first?.name ?? 'Seña',
      attempts: 0,
      correctCount: 0,
      totalSigns: _signs.length,
      user: HiveService.currentUser,
    );
    notifyListeners();

    await initialize();
  }

  Future<void> initialize() async {
    final hasPermission = await inferenceService.checkCameraPermission();
    if (!hasPermission) {
      _state = _state.copyWith(
        phase: PracticePhase.error,
        feedbackMessage:
            'Permisos de cámara denegados. Podés continuar navegando la teoría y la biblioteca.',
      );
      notifyListeners();
      return;
    }

    await cameraService.initialize();
    if (cameraService.hasError) {
      _state = _state.copyWith(
        phase: PracticePhase.error,
        feedbackMessage:
            '${cameraService.errorMessage}. Podés continuar navegando la teoría y la biblioteca.',
      );
      notifyListeners();
      return;
    }

    await inferenceService.loadModel();
    if (!inferenceService.isLoaded) {
      _logger.w('Modelo TFLite no disponible: modo conceptual activo.');
    }

    _startPreparation();
  }

  void _startPreparation() {
    _state = _state.copyWith(
      phase: PracticePhase.preparing,
      handDetected: false,
      lightingOk: true,
      isPositiveFeedback: false,
      feedbackMessage:
          'Coloca tu mano dentro del área indicada y espera el encuadre.',
      showGuide: true,
    );
    notifyListeners();

    // Pequeña pausa formal de "Preparación" antes de abrir la captura
    Future.delayed(const Duration(milliseconds: 800), () {
      if (_state.phase == PracticePhase.preparing) _startDetection();
    });
  }

  void _startDetection() {
    cameraService.startImageStream(_onFrame);
    _state = _state.copyWith(
      phase: PracticePhase.detecting,
      feedbackMessage: 'Buscando tu mano... Centra tu mano dentro del área indicada',
    );
    notifyListeners();
  }

  Future<void> _onFrame(CameraImage image) async {
    if (_isProcessing ||
        _state.phase == PracticePhase.evaluating ||
        _state.phase == PracticePhase.feedback ||
        _state.phase == PracticePhase.lessonFinished) {
      return;
    }

    // 1. Verificación de entorno (contraluz / ambiente oscuro)
    final lightingOk = await inferenceService.checkEnvironment(
      controller: cameraService.controller!,
      image: image,
    );
    if (!lightingOk) {
      _state = _state.copyWith(
        lightingOk: false,
        phase: PracticePhase.preparing,
        feedbackMessage:
            '✕ Necesita mejorar: la iluminación no es suficiente (contraluz u ambiente oscuro).',
      );
      notifyListeners();
      return;
    }

    // 2. Detección de la mano en el encuadre
    final handDetected = _simulateHandDetection();
    if (!handDetected) {
      _state = _state.copyWith(
        handDetected: false,
        phase: PracticePhase.detecting,
        feedbackMessage:
            'No pudimos detectar correctamente la mano. Centra tu mano dentro del área indicada',
      );
      notifyListeners();
      return;
    }

    // 3. Evaluación de la postura
    _state = _state.copyWith(
      handDetected: true,
      phase: PracticePhase.evaluating,
      feedbackMessage: 'Evaluando postura...',
    );
    notifyListeners();

    await _evaluateCurrentSign(image);
  }

  bool _simulateHandDetection() {
    // Reconocimiento conceptual (Fase 1). En Fase 2 aquí vive MediaPipe Hands.
    return _signs.isNotEmpty;
  }

  Future<void> _evaluateCurrentSign(CameraImage image) async {
    if (_isProcessing) return;
    _isProcessing = true;

    final user = HiveService.currentUser;
    final result = await inferenceService.processFrame(
      image: image,
      targetLabel: _state.currentSignId,
      userLevel: user.level,
    );

    _isProcessing = false;

    if (result == null) {
      _state = _state.copyWith(
        phase: PracticePhase.detecting,
        feedbackMessage: 'Reintentando detección...',
      );
      notifyListeners();
      return;
    }

    final attempts = _state.attempts + 1;

    if (result.isCorrect) {
      // Otorga +10 XP locales y guarda en Hive
      final updatedUser = await HiveService.addXp(AppConstants.xpPerCorrectSign);
      final newXCount = _state.correctCount + 1;

      _state = _state.copyWith(
        phase: PracticePhase.feedback,
        attempts: attempts,
        correctCount: newXCount,
        feedbackMessage: '¡Muy bien! +${AppConstants.xpPerCorrectSign} XP',
        isPositiveFeedback: true,
        user: updatedUser,
      );
      notifyListeners();

      _feedbackTimer?.cancel();
      _feedbackTimer = Timer(const Duration(milliseconds: 1600), () {
        if (newXCount >= _state.totalSigns) {
          _finishLesson();
        } else {
          _nextSign();
        }
      });
    } else {
      _state = _state.copyWith(
        phase: PracticePhase.feedback,
        attempts: attempts,
        feedbackMessage: result.feedback,
        isPositiveFeedback: false,
      );
      notifyListeners();

      _feedbackTimer?.cancel();
      _feedbackTimer = Timer(const Duration(milliseconds: 1800), () {
        if (_state.phase == PracticePhase.feedback) {
          _state = _state.copyWith(
            phase: PracticePhase.detecting,
            feedbackMessage: 'Intentá de nuevo. Revisá la guía y corregí la postura.',
          );
          notifyListeners();
        }
      });
    }
  }

  void _nextSign() {
    _signIndex = (_signIndex + 1).clamp(0, _signs.length - 1);
    final next = _signs[_signIndex];
    cameraService.stopImageStream();
    _state = _state.copyWith(
      currentSignId: next.id,
      signName: next.name,
      phase: PracticePhase.preparing,
      handDetected: false,
      feedbackMessage: 'Siguiente seña: ${next.name}. Preparate para el encuadre.',
      isPositiveFeedback: false,
    );
    notifyListeners();
    _startPreparation();
  }

  Future<void> _finishLesson() async {
    cameraService.stopImageStream();

    String? lessonId;
    if (_signs.isNotEmpty) {
      final matches = HiveService.lessons
          .where((l) => l.signIds.contains(_signs.first.id));
      if (matches.isNotEmpty) lessonId = matches.first.id;
    }
    if (lessonId != null) {
      await HiveService.markLessonCompleted(lessonId);
    }

    _state = _state.copyWith(
      phase: PracticePhase.lessonFinished,
      showSummary: true,
      feedbackMessage: '¡Lección completada!',
      user: HiveService.currentUser,
    );
    notifyListeners();
  }

  // ─── Acciones de la UI ───
  void retry() {
    _feedbackTimer?.cancel();
    _state = _state.copyWith(
      phase: PracticePhase.preparing,
      feedbackMessage: 'Coloca tu mano dentro del área indicada.',
    );
    notifyListeners();
    _startPreparation();
  }

  void showGuideToggle() {
    _state = _state.copyWith(showGuide: !_state.showGuide);
    notifyListeners();
  }

  void closeSummary() {
    _state = _state.copyWith(showSummary: false, phase: PracticePhase.completed);
    notifyListeners();
  }

  Future<void> stop() async {
    _feedbackTimer?.cancel();
    cameraService.stopImageStream();
  }

  @override
  void dispose() {
    _feedbackTimer?.cancel();
    cameraService.stopImageStream();
    super.dispose();
  }
}

              child: Text(

                style: const TextStyle(color: Colors.white, fontSize: 14),
                textAlign: TextAlign.center,
              ),
            ),
          ),
        ),
      ],
    );
  }

  Color _getBorderColor() {
    if (state.phase == PracticePhase.error) return Colors.red;
    if (state.phase == PracticePhase.lessonFinished ||
        state.isPositiveFeedback) {
      return Colors.green;
    }
    if (state.phase == PracticePhase.evaluating && state.handDetected) {
      return Colors.blue;
    }
    if (!state.lightingOk) return Colors.orange;
    return Colors.white;
  }

  String _getStatusText() {
    if (!state.lightingOk) return 'Iluminación insuficiente o contraluz';
    switch (state.phase) {
      case PracticePhase.preparing:
        return 'PREPARACIÓN: Coloca tu mano dentro del área';
      case PracticePhase.detecting:
        return 'Buscando mano...';
      case PracticePhase.evaluating:
        return 'Evaluando postura...';
      case PracticePhase.feedback:
        return state.isPositiveFeedback
            ? '¡Postura correcta!'
            : 'Revisá la guía y corregí';
      case PracticePhase.lessonFinished:
        return '¡Lección completada!';
      case PracticePhase.completed:
        return 'Listo';
      case PracticePhase.error:
        return 'Sin cámara';
    }
  }
}

@HiveType(typeId: 4)
class ProgressModel extends Equatable {
  const ProgressModel({
    required this.userId,
    required this.completedSignIds,
    required this.streakDays,
    required this.lastPracticeDate,
    required this.totalPracticeTimeSeconds,
    required this.correctAttempts,
    required this.totalAttempts,
  });

  @HiveField(0)
  final String userId;
  @HiveField(1)
  final List<String> completedSignIds;
  @HiveField(2)
  final int streakDays;
  @HiveField(3)
  final DateTime? lastPracticeDate;
  @HiveField(4)
  final int totalPracticeTimeSeconds;
  @HiveField(5)
  final int correctAttempts;
  @HiveField(6)
  final int totalAttempts;

  double get accuracy => totalAttempts == 0 ? 0 : correctAttempts / totalAttempts;

  ProgressModel copyWith({
    String? userId,
    List<String>? completedSignIds,
    int? streakDays,
    DateTime? lastPracticeDate,
    int? totalPracticeTimeSeconds,
    int? correctAttempts,
    int? totalAttempts,
  }) {
    return ProgressModel(
      userId: userId ?? this.userId,
      completedSignIds: completedSignIds ?? this.completedSignIds,
      streakDays: streakDays ?? this.streakDays,
      lastPracticeDate: lastPracticeDate ?? this.lastPracticeDate,
      totalPracticeTimeSeconds: totalPracticeTimeSeconds ?? this.totalPracticeTimeSeconds,
      correctAttempts: correctAttempts ?? this.correctAttempts,
      totalAttempts: totalAttempts ?? this.totalAttempts,
    );
  }

  @override

        userId,
        completedSignIds,
        streakDays,
        lastPracticeDate,

import 'package:flutter/material.dart';
import 'package:flutter/services.dart';
import 'package:senasgo/core/services/device_performance.dart';

/// Avatar interactivo (Profesor) que demuestra las señas.
///
/// - Modo principal: Rive (assets/rive/professor.riv) cuando el dispositivo
///   tiene memoria suficiente.
/// - Fallback Visual: si el dispositivo tiene poca RAM (o el asset no existe),
///   transiciona automáticamente a una animación 2D ligera para evitar
///   bloqueos o degradación del rendimiento.
class ProfessorAvatar extends StatefulWidget {
  const ProfessorAvatar({
    super.key,
    this.size = 160,
    this.animate = true,
    this.signId,
  });

  final double size;

  /// Activa la demostración de la seña.
  final bool animate;

  /// Seña a demostrar (usada por el state machine de Rive / por la pose 2D).
  final String? signId;

  @override
  State<ProfessorAvatar> createState() => _ProfessorAvatarState();
}

class _ProfessorAvatarState extends State<ProfessorAvatar>
    with SingleTickerProviderStateMixin {
  bool? _riveAvailable; // null = cargando
  late final AnimationController _controller;
  late final Animation<double> _armAngle;

  @override
  void initState() {
    super.initState();
    _controller = AnimationController(
      vsync: this,
      duration: const Duration(milliseconds: 1400),
    );
    _armAngle = Tween<double>(begin: -0.35, end: 0.55).animate(
      CurvedAnimation(parent: _controller, curve: Curves.easeInOut),
    );
    if (widget.animate) _controller.repeat(reverse: true);
    _probeRive();
  }

  Future<void> _probeRive() async {
    if (DevicePerformance.isLowMemory) {
      // Fallback inmediato en dispositivos con poca RAM
      if (mounted) setState(() => _riveAvailable = false);
      return;
    }
    try {
      final asset = widget.signId == null
          ? 'assets/rive/professor.riv'
          : 'assets/rive/${widget.signId}.riv';
      await rootBundle.load(asset);
      if (mounted) setState(() => _riveAvailable = true);
    } catch (_) {
      // Asset inexistente -> fallback 2D
      if (mounted) setState(() => _riveAvailable = false);
    }
  }

  @override
  void didUpdateWidget(covariant ProfessorAvatar oldWidget) {
    super.didUpdateWidget(oldWidget);
    if (widget.animate && !_controller.isAnimating) {
      _controller.repeat(reverse: true);
    } else if (!widget.animate) {
      _controller.stop();
    }
  }

  @override
  void dispose() {
    _controller.dispose();
    super.dispose();
  }

  @override
  Widget build(BuildContext context) {
    if (_riveAvailable == null) {
      return SizedBox(
        width: widget.size,
        height: widget.size,
        child: const Center(
          child: SizedBox(
            width: 24,
            height: 24,
            child: CircularProgressIndicator(strokeWidth: 2),
          ),
        ),
      );
    }

    // Transición suave entre modos (evita "flash")
    return AnimatedSwitcher(
      duration: const Duration(milliseconds: 350),
      child: _riveAvailable == true
          ? const Center(key: ValueKey('rive'))
          : _TwoDProfessor(
              key: const ValueKey('2d'),
              size: widget.size,
              animation: _armAngle,
            ),
    );
  }
}

/// Fallback 2D: profesor animado dibujado con Flutter (sin assets pesados).
/// Rendimiento mínimo en cualquier dispositivo.
class _TwoDProfessor extends StatelessWidget {
  const _TwoDProfessor({
    super.key,
    required this.size,
    required this.animation,
  });

  final double size;
  final Animation<double> animation;

  @override
  Widget build(BuildContext context) {
    final accent = Theme.of(context).colorScheme.primary;
    return AnimatedBuilder(
      animation: animation,
      builder: (context, _) {
        return SizedBox(
          width: size,
          height: size,
          child: CustomPaint(
            painter: _ProfessorPainter(
              armAngle: animation.value,
              accent: accent,
            ),
          ),
        );
      },
    );
  }
}

class _ProfessorPainter extends CustomPainter {
  _ProfessorPainter({required this.armAngle, required this.accent});

  final double armAngle;
  final Color accent;

  @override
  void paint(Canvas canvas, Size size) {
    final w = size.width;
    final h = size.height;
    final center = Offset(w * 0.5, h * 0.5);

    final bodyPaint = Paint()..color = accent;
    final skinPaint = Paint()..color = const Color(0xFFFFCBA4);
    final darkPaint = Paint()..color = const Color(0xFF333355);
    final stroke = Paint()
      ..color = const Color(0xFF222244)
      ..style = PaintingStyle.stroke
      ..strokeWidth = 2;

    // Cuerpo
    canvas.drawRRect(
      RRect.fromRectAndRadius(
        Rect.fromCenter(center: Offset(center.dx, h * 0.72), width: w * 0.44, height: h * 0.34),
        const Radius.circular(18),
      ),
      bodyPaint,
    );

    // Cabeza
    canvas.drawCircle(Offset(center.dx, h * 0.34), w * 0.17, skinPaint);
    canvas.drawCircle(Offset(center.dx, h * 0.34), w * 0.17, stroke);

    // Pelo
    canvas.drawArc(
      Rect.fromCircle(center: Offset(center.dx, h * 0.34), radius: w * 0.17),
      3.1416,
      3.1416,
      true,
      darkPaint,
    );

    // Brazo fijo (izquierdo)
    final shoulderL = Offset(center.dx - w * 0.2, h * 0.62);
    canvas.drawLine(
      shoulderL,
      Offset(shoulderL.dx - w * 0.1, shoulderL.dy + h * 0.16),
      Paint()
        ..color = accent
        ..strokeWidth = 10
        ..strokeCap = StrokeCap.round,
    );

    // Brazo demostrador (derecho) - rota con la animación
    final shoulderR = Offset(center.dx + w * 0.2, h * 0.62);
    final angle = armAngle;
    final len = h * 0.26;
    final hand = Offset(
      shoulderR.dx + len * (0.45 * angle + 0.9),
      shoulderR.dy - len * (0.8 - angle),
    );
    canvas.drawLine(
      shoulderR,
      hand,
      Paint()
        ..color = accent
        ..strokeWidth = 10
        ..strokeCap = StrokeCap.round,
    );

    // Mano (configuración manual simulada)
    canvas.drawCircle(hand, w * 0.055, skinPaint);
    canvas.drawCircle(hand, w * 0.055, stroke);

    // Dedos extendidos (seña)
    for (var i = -1; i <= 1; i++) {
      canvas.drawLine(
        hand + Offset(i * 6.0, -4),
        hand + Offset(i * 9.0, -16 - i.abs() * 3),
        Paint()
          ..color = const Color(0xFF222244)
          ..strokeWidth = 3
          ..strokeCap = StrokeCap.round,
      );
    }
  }

  @override
  bool shouldRepaint(covariant _ProfessorPainter oldDelegate) =>
      oldDelegate.armAngle != armAngle || oldDelegate.accent != accent;
}
